29    async fn fetch_quota(&self, key: &str) -> Result<ProviderResult> {
30        // Project is optional for the summary endpoint (empty body works),
31        // but prefer the resolved workspace/default project when known.
32        let project = resolve_project_id();
33        let body = match project.as_deref() {
34            Some(p) if !p.is_empty() => serde_json::json!({ "project": p }),
35            _ => serde_json::json!({}),
36        };
37
38        // cloudcode-pa rejects calls without a recognized client UA (403
39        // PERMISSION_DENIED). The agy binary identifies as "antigravity".
40        let resp = self
41            .http
42            .post("https://cloudcode-pa.googleapis.com/v1internal:retrieveUserQuotaSummary")
43            .header("Authorization", format!("Bearer {}", key))
44            .header("Content-Type", "application/json")
45            .header("User-Agent", "antigravity")
46            .json(&body)
47            .send()
48            .await?;
49
50        let status = resp.status();
51        let body: serde_json::Value = resp.json().await?;
52
53        if status.as_u16() == 401 || status.as_u16() == 403 {
54            return Ok(ProviderResult {
55                kind: ProviderKind::Antigravity,
56                status: ProviderStatus::AuthRequired,
57                fetched_at: Utc::now(),
58                raw_response: Some(body),
59                auth_source: None,
60                cached_at: None,
61            instance_key: None,
62            profile_id: None,
63            card_title: None,
64            });
65        }
66
67        if !status.is_success() {
68            let msg = body
69                .get("error")
70                .and_then(|v| v.get("message"))
71                .and_then(|v| v.as_str())
72                .or_else(|| body.get("message").and_then(|v| v.as_str()))
73                .unwrap_or("unknown error");
74            return Ok(ProviderResult {
75                kind: ProviderKind::Antigravity,
76                status: ProviderStatus::Unavailable {
77                    info: crate::providers::UnavailableInfo {
78                        reason: msg.to_string(),
79                        console_url: Some("https://antigravity.google".into()),
80                    },
81                },
82                fetched_at: Utc::now(),
83                raw_response: Some(body),
84                auth_source: None,
85                cached_at: None,
86            instance_key: None,
87            profile_id: None,
88            card_title: None,
89            });
90        }
91
92        let quota = parse_quota(&body)?;
93        Ok(ProviderResult {
94            kind: ProviderKind::Antigravity,
95            status: ProviderStatus::Available { quota },
96            fetched_at: Utc::now(),
97            raw_response: Some(body),
98            auth_source: None,
99            cached_at: None,
100            instance_key: None,
101            profile_id: None,
102            card_title: None,
103        })
104    }

107pub(crate) fn resolve_project_id() -> Option<String> {
108    if let Ok(project) = std::env::var("GOOGLE_CLOUD_PROJECT") {
109        let project = project.trim();
110        if !project.is_empty() {
111            return Some(project.to_string());
112        }
113    }
114    if let Ok(project) = std::env::var("GOOGLE_CLOUD_PROJECT_ID") {
115        let project = project.trim();
116        if !project.is_empty() {
117            return Some(project.to_string());
118        }
119    }
120
121    // agy default project cache
122    if let Some(home) = gemini_home_dir() {
123        let default = home
124            .join("antigravity-cli/cache/default_project_id.txt");
125        if let Ok(s) = std::fs::read_to_string(&default) {
126            let s = s.trim();
127            if !s.is_empty() {
128                return Some(s.to_string());
129            }
130        }
131    }
132
133    resolve_project_id_for_workspace(&std::env::current_dir().ok()?, &gemini_home_dir()?)
134}

136pub(crate) fn resolve_project_id_for_workspace(

187fn derive_project_slug(base_name: &str, existing_projects: &BTreeMap<String, String>) -> String {
188    let slug = slugify(base_name);
189    let existing_ids: BTreeSet<&str> = existing_projects.values().map(String::as_str).collect();
190    if !existing_ids.contains(slug.as_str()) {
191        return slug;
192    }
193
194    let mut counter = 1usize;
195    loop {
196        let candidate = format!("{}-{}", slug, counter);
197        if !existing_ids.contains(candidate.as_str()) {
198            return candidate;
199        }
200        counter += 1;
201    }
202}

204fn slugify(text: &str) -> String {
205    let slug = text
206        .to_lowercase()
207        .chars()
208        .map(|c| if c.is_ascii_alphanumeric() { c } else { '-' })
209        .collect::<String>();
210    let slug = slug.trim_matches('-').to_string();
211    let mut collapsed = String::new();
212    let mut last_dash = false;
213    for ch in slug.chars() {
214        if ch == '-' {
215            if !last_dash {
216                collapsed.push(ch);
217            }
218            last_dash = true;
219        } else {
220            collapsed.push(ch);
221            last_dash = false;
222        }
223    }
224    if collapsed.is_empty() {
225        "project".to_string()
226    } else {
227        collapsed
228    }
229}

266/// Parse a `/v1internal:retrieveUserQuotaSummary` (or compatible) body.
267///
268/// Emits one `QuotaWindow` per bucket, with `window_type` shaped as
269/// `{period}/{group}` (e.g. `5h/gemini`, `7d/3p`) so the TUI clusters
270/// by period and labels by group — matching the agy CLI's 2×(5h+7d) grid.
271pub(crate) fn parse_quota(body: &serde_json::Value) -> Result<ProviderQuota> {
272    // Prefer the summary shape; fall back to the older per-model buckets
273    // list (`retrieveUserQuota`) so cached/legacy payloads still parse.
274    if body.get("groups").is_some() || looks_like_summary_buckets(body) {
275        return parse_summary_quota(body);
276    }
277    parse_legacy_model_quota(body)
278}

280fn looks_like_summary_buckets(body: &serde_json::Value) -> bool {
281    body.get("buckets")
282        .and_then(|b| b.as_array())
283        .and_then(|arr| arr.first())
284        .map(|b| b.get("bucketId").is_some() || b.get("window").is_some())
285        .unwrap_or(false)
286}

288fn parse_summary_quota(body: &serde_json::Value) -> Result<ProviderQuota> {
289    let resp: QuotaSummaryResponse = serde_json::from_value(body.clone())
290        .map_err(|e| crate::Error::Provider(format!("parse error: {}", e)))?;
291
292    let mut windows = Vec::new();
293
294    if !resp.groups.is_empty() {
295        for group in &resp.groups {
296            let group_label = group_slug(group);
297            for bucket in &group.buckets {
298                if let Some(w) = summary_bucket_to_window(bucket, &group_label) {
299                    windows.push(w);
300                }
301            }
302        }
303    } else {
304        // Flat buckets with bucketId/window — no group wrapper.
305        for bucket in &resp.buckets {
306            let label = bucket
307                .bucket_id
308                .as_deref()
309                .and_then(|id| id.rsplit_once('-').map(|(p, _)| p.to_string()))
310                .unwrap_or_else(|| "group".into());
311            if let Some(w) = summary_bucket_to_window(bucket, &label) {
312                windows.push(w);
313            }
314        }
315    }
316
317    let plan_name = if windows.is_empty() {
318        "Antigravity (no quota data)".to_string()
319    } else {
320        "Antigravity".to_string()
321    };
322
323    Ok(ProviderQuota {
324        plan_name,
325        windows,
326        unlimited: false,
327        banked_resets: None,
328    })
329}

331fn group_slug(group: &QuotaSummaryGroup) -> String {
332    // Prefer stable bucketId prefix (gemini-weekly → gemini, 3p-5h → 3p).
333    for bucket in &group.buckets {
334        if let Some(id) = bucket.bucket_id.as_deref() {
335            if let Some((prefix, rest)) = id.rsplit_once('-') {
336                if matches!(rest, "weekly" | "5h" | "7d" | "wk") && !prefix.is_empty() {
337                    return prefix.to_string();
338                }
339            }
340        }
341    }
342    // Fallback: first word of displayName lowercased.
343    group
344        .display_name
345        .as_deref()
346        .and_then(|s| s.split_whitespace().next())
347        .map(|s| s.to_ascii_lowercase())
348        .filter(|s| !s.is_empty())
349        .unwrap_or_else(|| "group".into())
350}

352fn summary_bucket_to_window(bucket: &QuotaSummaryBucket, group: &str) -> Option<QuotaWindow> {
353    let period_key = normalize_window_key(
354        bucket
355            .window
356            .as_deref()
357            .or_else(|| {
358                // Infer from bucketId suffix when `window` is absent.
359                bucket
360                    .bucket_id
361                    .as_deref()
362                    .and_then(|id| id.rsplit_once('-').map(|(_, rest)| rest))
363            })
364            .unwrap_or("window"),
365    );
366
367    let (used, limit, remaining) = fraction_to_counts(
368        bucket.remaining_fraction,
369        bucket.remaining_amount.as_deref(),
370    );
371
372    let reset_at = bucket.reset_time.as_ref().and_then(|t| {
373        DateTime::parse_from_rfc3339(t)
374            .map(|dt| dt.with_timezone(&Utc))
375            .ok()
376    });
377
378    let period_seconds = match period_key.as_str() {
379        "5h" => Some(5 * 3600),
380        "7d" => Some(7 * 86400),
381        _ => None,
382    };
383
384    // Prefer a short display-friendly group token; fall back to group slug.
385    let _ = bucket.display_name.as_ref(); // reserved for future richer labels
386
387    Some(QuotaWindow {
388        window_type: format!("{period_key}/{group}"),
389        used,
390        limit,
391        remaining,
392        reset_at,
393        period_seconds,
394    })
395}

397fn normalize_window_key(raw: &str) -> String {
398    match raw.trim().to_ascii_lowercase().as_str() {
399        "weekly" | "week" | "wk" | "7d" | "168h" => "7d".into(),
400        "5h" | "five_hour" | "five-hour" | "5hour" | "300m" => "5h".into(),
401        "" => "window".into(),
402        other => other.to_string(),
403    }
404}

406/// Convert remainingFraction (0..1) and/or remainingAmount into used/limit/remaining.
407/// Percentage scale (limit=100) when only a fraction is present — matches
408/// how the agy CLI and other quotas bars render utilization.
409fn fraction_to_counts(
410    remaining_fraction: Option<f64>,
411    remaining_amount: Option<&str>,
412) -> (i64, i64, i64) {
413    match (
414        remaining_amount.and_then(|s| s.parse::<i64>().ok()),
415        remaining_fraction,
416    ) {
417        (Some(remaining), Some(fraction)) if fraction > 0.0 => {
418            let limit = (remaining as f64 / fraction).round() as i64;
419            let used = limit.saturating_sub(remaining);
420            (used, limit, remaining)
421        }
422        (Some(remaining), _) => (0, remaining, remaining),
423        (None, Some(fraction)) => {
424            let limit = 100i64;
425            // Clamp to [0, 100]; round half-up for the remaining percent.
426            let remaining = (fraction.clamp(0.0, 1.0) * limit as f64).round() as i64;
427            let used = limit.saturating_sub(remaining);
428            (used, limit, remaining)
429        }
430        (None, None) => (0, 0, 0),
431    }
432}

456fn parse_legacy_model_quota(body: &serde_json::Value) -> Result<ProviderQuota> {
457    let resp: LegacyQuotaResponse = serde_json::from_value(body.clone())
458        .map_err(|e| crate::Error::Provider(format!("parse error: {}", e)))?;
459
460    let mut windows: Vec<QuotaWindow> = Vec::new();
461
462    for bucket in &resp.buckets {
463        let (used, limit, remaining) = fraction_to_counts(
464            bucket.remaining_fraction,
465            bucket.remaining_amount.as_deref(),
466        );
467
468        let window_type = match (&bucket.token_type, &bucket.model_id) {
469            (Some(tt), Some(mid)) => format!("{}_{}", tt.to_uppercase(), mid),
470            (Some(tt), None) => tt.to_uppercase(),
471            (None, Some(mid)) => mid.clone(),
472            (None, None) => "unknown".to_string(),
473        };
474
475        let reset_at = bucket.reset_time.as_ref().and_then(|t| {
476            DateTime::parse_from_rfc3339(t)
477                .map(|dt| dt.with_timezone(&Utc))
478                .ok()
479        });
480
481        windows.push(QuotaWindow {
482            window_type,
483            used,
484            limit,
485            remaining,
486            reset_at,
487            period_seconds: None,
488        });
489    }
490
491    let plan_name = if windows.is_empty() {
492        "Antigravity (no quota data)".to_string()
493    } else {
494        "Antigravity".to_string()
495    };
496
497    Ok(ProviderQuota {
498        plan_name,
499        windows,
500        unlimited: false,
501        banked_resets: None,
502    })
503}
